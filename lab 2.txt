class Product:
    def __init__(self, name, price, stock):
        self._name = name
        self._price = price
        self._stock = stock

    def get_total_price(self, quantity):
        if quantity <= 0:
            raise ValueError("Quantity must be greater than 0")
        if quantity > self._stock:
            raise ValueError("Not enough stock")
        return self._price * quantity

    def display_info(self):
        print(f"Product: {self._name}, Price: {self._price} €, Stock: {self._stock}")

class Book(Product):
    def __init__(self, name, price, stock, author):
        super().__init__(name, price, stock)
        self._author = author

    def get_total_price(self, quantity):
        base_price = super().get_total_price(quantity)
        if quantity >= 3:
            return base_price * 0.9  
        return base_price

    def display_info(self):
        super().display_info()
        print(f"Author: {self._author}")

class Electronics(Product):
    def __init__(self, name, price, stock, warranty_years):
        super().__init__(name, price, stock)
        self._warranty_years = warranty_years

    def get_total_price(self, quantity):
        base_price = super().get_total_price(quantity)
        warranty_cost = self._warranty_years * 10
        return base_price + warranty_cost

    def display_info(self):
        super().display_info()
        print(f"Warranty: {self._warranty_years} years")
        
class CartItem:
    
    def __init__(self, product, quantity):
        self.product = product
        self.quantity = quantity

    def get_subtotal(self):
        return self.product.get_total_price(self.quantity)

class ShoppingCart:
    
    def __init__(self):
        self._items = []  

    def add_product(self, product, quantity):
        new_item = CartItem(product, quantity)
        self._items.append(new_item)

    def remove_product(self, product_name):
        self._items = [item for item in self._items if item.product._name != product_name]

    def get_total(self):
        return sum(item.get_subtotal() for item in self._items)

    def display_cart(self):
        if not self._items:
            print("Cart is empty.")
            return
        print("--- Shopping Cart Items ---")
        for item in self._items:
            print(f"- {item.product._name} x{item.quantity}: {item.get_subtotal()} €")
        print(f"Total: {self.get_total()} €")

    def clear_cart(self):
        self._items = []

class Customer:
    def __init__(self, name, email):
        self._name = name
        self._email = email
        self._cart = ShoppingCart()

    def shop(self, product, quantity):
        self._cart.add_product(product, quantity)

    def view_cart(self):
        print(f"\nCustomer: {self._name}'s Cart")
        self._cart.display_cart()

    def checkout(self):
        total = self._cart.get_total()
        print(f"\nProceeding to checkout for {self._name}...")
        print(f"Final amount paid: {total} €")
        self._cart.clear_cart()
        return total

class CartManager:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._carts = {}
        return cls._instance

    def get_cart(self, customer_id):
        if customer_id not in self._carts:
            self._carts[customer_id] = ShoppingCart()
        return self._carts[customer_id]

    def get_all_carts_total(self):
        return sum(cart.get_total() for cart in self._carts.values())

class Customer:
    def __init__(self, name, email, customer_id="default_id"): 
        self._name = name
        self._email = email
        self._id = customer_id
        self._manager = CartManager()
        self._cart = self._manager.get_cart(self._id)

    def shop(self, product, quantity):
        self._cart.add_product(product, quantity)

    def view_cart(self):
        print(f"\nCustomer: {self._name}'s Cart (ID: {self._id})")
        self._cart.display_cart()

    def checkout(self):
        total = self._cart.get_total()
        print(f"\nProceeding to checkout for {self._name}...")
        print(f"Final amount paid: {total} €")
        self._cart.clear_cart()
        return total
